function toggleWordChip(chip, word) { playSound('click'); resetCheckState(); const answerZone = document.getElementById('answer-zone'); const wordPool = document.getElementById('word-pool'); if (chip.parentElement === wordPool) { // ให้พูดเฉพาะตอนที่กดเลือกคำศัพท์เข้าไปในช่องตอบ (Answer Zone) speakJapaneseText(word); answerZone.appendChild(chip); userSelectedSequence.push(word); } else { // ตอนดึงคำศัพท์กลับมาที่ช่องเดิม (Word Pool) ไม่ต้องพูด wordPool.appendChild(chip); const index = userSelectedSequence.indexOf(word); if (index > -1) userSelectedSequence.splice(index, 1); } }
